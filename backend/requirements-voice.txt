#     pip install -r backend/requirements-voice.txt
#     python -m spacy download en_core_web_sm

# Kokoro pulls torch and transformers, and reaches spaCy through misaki for
# grapheme-to-phoneme. misaki does not declare spaCy in its metadata, so it is
# pinned here explicitly - a reverse-dependency check cannot see it, and
# removing it breaks speech with "No module named 'spacy'" at synthesis time.
#
# en_core_web_sm is fetched through spaCy's downloader rather than pinned as a
# GitHub URL, because that URL fails the entire install when a connection drops.
#

# import and disables speech while chat continues - see voice/providers/kokoro.py.
